117 void computePowerFlow(std::map<std::string, double>& data,
118 std::map<std::string, double>& globalParams) const override
119 {
120 for (auto& [key, value] : element_OPF_info)
121 data[key] = value; // Copy OPF info to branch data
122
123 // Match the OPF converter equivalent to the MMC plant: no AC filter /
124 // transformer, and rc+jxc = R_eq + jω L_eq with L_eq = L_arm/2 + L_reactor.
125 // MatACDC row defaults (bf=0.0887 pu, LossA=1.103 MW, xc=0.164 pu) would
126 // otherwise produce a PCC (V,P,Q,Pdc) that is not an MMC equilibrium.
127 if (!element_OPF_info.count("bf"))
128 data["bf"] = 0.0;
129 if (!element_OPF_info.count("rtf"))
130 data["rtf"] = 0.0;
131 if (!element_OPF_info.count("xtf"))
132 data["xtf"] = 0.0;
133 if (!element_OPF_info.count("LossA"))
134 data["LossA"] = 0.0;
135 if (!element_OPF_info.count("LossB"))
136 data["LossB"] = 0.0;
137 if (!element_OPF_info.count("LossCrec"))
138 data["LossCrec"] = 0.0;
139 if (!element_OPF_info.count("LossCinv"))
140 data["LossCinv"] = 0.0;
141 if (!element_OPF_info.count("xc"))
142 data["xc"] = globalParams["omega"] * L_eq / globalParams["ACZbase"];
143 if (!element_OPF_info.count("rc"))
144 data["rc"] = R_eq / globalParams["ACZbase"];
145 // MatACDC P_G/Q_G: OPF enforces pn = -P_G and qs = -Q_G.
146 // P_G/Q_G are MMC machine powers in MW (report Park Q = 1.5(Vq Id − Vd Iq)).
147 data["P_g"] = P / 1e6;
148 data["Q_g"] = Q / 1e6;
149 data["Vtar"] = V_dc / 1e3 / globalParams["DCbaseKV"]; // Seting of DC v-control value
150
151 data["gridac"] = (int)element_location[2] - '0'; // AC grid number
152
153 // DC side type_dc(1 = constant DC power control (i.e. active power), 2 = constant DC voltage control, 3 = DC droop control)
154 if (controls.count("active_power")) {
155 if (!element_OPF_info.count("type_dc"))
156 data["type_dc"] = 1;
157 }
158 else if (controls.count("dc_voltage")) {
159 if (!element_OPF_info.count("type_dc"))
160 data["type_dc"] = 2;
161 }
162 else if (controls.count("droop")) {
163 if (!element_OPF_info.count("type_dc"))
164 data["type_dc"] = 3;
165 }
166
167 // AC side type_ac matches Powerflow_solver / MatACDC: 1 = Q, 2 = Vac.
168 if (controls.count("ac_voltage")) {
169 if (!element_OPF_info.count("type_ac"))
170 data["type_ac"] = 2;
171 }
172 else if (controls.count("reactive_power")) {
173 if (!element_OPF_info.count("type_ac"))
174 data["type_ac"] = 1;
175 }
176
177 if (element_OPF_info.count("type_dc"))
178 data["type_dc"] = element_OPF_info.at("type_dc");
179 if (element_OPF_info.count("type_ac"))
180 data["type_ac"] = element_OPF_info.at("type_ac");
181 }
